67 // Represents a character coordinate from the user's point of view,
68 // i. e. consider an uniform grid (assuming fixed-width font) on the
69 // screen as it is rendered, and each cell has its own coordinate, starting from 0.
70 // Tabs are counted as [1..mTabSize] count empty spaces, depending on
71 // how many space is necessary to reach the next tab stop.
72 // For example, coordinate (1, 5) represents the character 'B' in a line "\tABC", when mTabSize = 4,
73 // because it is rendered as " ABC" on the screen.
74 struct Coordinates
75 {
76 int mLine, mColumn;
77 Coordinates() : mLine(0), mColumn(0) {}
78 Coordinates(int aLine, int aColumn) : mLine(aLine), mColumn(aColumn)
79 {
80 assert(aLine >= 0);
81 assert(aColumn >= 0);
82 }
83 static Coordinates Invalid() { static Coordinates invalid(-1, -1); return invalid; }
84
85 bool operator ==(const Coordinates& o) const
86 {
87 return
88 mLine == o.mLine &&
89 mColumn == o.mColumn;
90 }
91
92 bool operator !=(const Coordinates& o) const
93 {
94 return
95 mLine != o.mLine ||
96 mColumn != o.mColumn;
97 }
98
99 bool operator <(const Coordinates& o) const
100 {
101 if (mLine != o.mLine)
102 return mLine < o.mLine;
103 return mColumn < o.mColumn;
104 }
105
106 bool operator >(const Coordinates& o) const
107 {
108 if (mLine != o.mLine)
109 return mLine > o.mLine;
110 return mColumn > o.mColumn;
111 }
112
