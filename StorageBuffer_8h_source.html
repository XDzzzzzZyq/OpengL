242template <typename T>
243void StorageBuffer::ReadStorageBuffer(std::vector<T>& tar, GLuint _offset)
244{
245 BindBuffer();
246 void* dataPtr = glMapBuffer(GL_SHADER_STORAGE_BUFFER, GL_READ_ONLY);
247
248 if (dataPtr == nullptr)
249 return;
250
251 /* | -- Offset -- | ----------- Target -----------|
252 * |------------- Whole Data Range -------------| */
253
254 if (tar.size() == 0) {
255 GLint bufferSize;
256 glGetBufferParameteriv(GL_SHADER_STORAGE_BUFFER, GL_BUFFER_SIZE, &bufferSize);
257 tar.resize(bufferSize - _offset);
258 }
259
260 std::memcpy(tar.data(), static_cast<char*>(dataPtr) + _offset, tar.size() * sizeof(T));
261
262 glUnmapBuffer(GL_SHADER_STORAGE_BUFFER);
263 UnbindBuffer();
264}
